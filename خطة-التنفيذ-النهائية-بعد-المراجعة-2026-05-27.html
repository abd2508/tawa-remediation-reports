٢. منهج المراجعة الدقيقة

كل بَند في هذه الخطة مَرَّ بـ5 خطوات تَحقُّق:

  1. قراءة الكود الفعلي — لم أَعتمد على تقرير الإيجنت وَحده، فَتحتُ كل ملف وقَرأت السطور المُحدَّدة بنفسي
  2. تَحديد الأعراض الجانبية — ما الملفات الأخرى التي تَستخدم نفس النمط؟ ما الـtest الذي يَكسر؟ ما السلوك الذي يَنخرم؟
  3. محاكاة ذهنية — ماذا يَحدث للمستخدم العادي أثناء التَّعديل + بَعده؟
  4. أسئلة المستخدم الطبيعي — إيجنت مُستقلّ يَلعب 4 أدوار يَطرح أسئلة لاكتشاف الفجوات
  5. اكتشاف false-positives — مثال: تقرير OPS قال 9 confirm()، التَّحقُّق أَثبت 8 (الـ9 كان دالّة محلية اسمها confirm في OpsDriverSettlements، ليس window.confirm)
📜 الـfalse-positives التي اكتشَفتُها في المراجعة
  • O-M3 (8 بدل 9): OpsDriverSettlements.tsx:100 يَستخدم confirm(s.id) — لكنّ confirm هنا دالّة محليّة لتأكيد الاستلام، ليس window.confirm (سَطر 105 يَستخدم setConfirmFor). أُقصِيَ من قائمة الإصلاحات
  • C-M3 أبسط ممّا قِيل: backend يَحترم الـflag الحالي (false) — التَّعديل المَطلوب هو فقط معالجة 422 OTP_REQUIRED صَريحاً، ليس feature flag dynamic
  • P-M1 أَدقّ: ProviderSettings ليس له ملف اختبار (تأكَّدتُ) → الإصلاح آمن، لا اختبارات تَنكسر
  • OpsDocumentInbox object_key: الحَقل مَوجود في DocumentInboxItem type (سَطر 686 of api/operations.ts) → AuthedImage يُضاف بـ4 أَسطر فقط

import type { UserRole } from "../api/types";

export const PROVIDER_ALL = ["PROVIDER_ADMIN", "PROVIDER_STAFF"] as const;
export const PROVIDER_ADMIN_ONLY = ["PROVIDER_ADMIN"] as const;

export const PROVIDER_NAV_ACCESS: Record<string, readonly UserRole[]> = {
  "/provider-portal":            PROVIDER_ALL,
  "/provider-portal/orders":     PROVIDER_ALL,
  "/provider-portal/catalog":    PROVIDER_ADMIN_ONLY,  // catalog edit = admin
  "/provider-portal/promotions": PROVIDER_ADMIN_ONLY,  // ← الإصلاح
  "/provider-portal/history":    PROVIDER_ALL,
  "/provider-portal/earnings":   PROVIDER_ALL,
  "/provider-portal/settlement": PROVIDER_ADMIN_ONLY,  // ← الإصلاح
  "/provider-portal/settings":   PROVIDER_ALL,
};

export function canAccessProviderPath(path: string, role: UserRole | undefined): boolean {
  if (!role) return false;
  const allowed = PROVIDER_NAV_ACCESS[path];
  if (!allowed) return true;  // fail-open للـnav
  return allowed.includes(role);
}

{canAccessProviderPath("/provider-portal/settlement", user?.role) && (

const respondSubMutation = useMutation({
  mutationFn: ({ subOfferId, accept }) =>
    ordersApi.respondSub(subOfferId, accept),
  onMutate: async ({ subOfferId }) => {
    // إلغاء أيّ refetch جارٍ
    await queryClient.cancelQueries({ queryKey: ["sub-offers", id] });
    // snapshot للقيمة القديمة (للـrollback)
    const previous = queryClient.getQueryData(["sub-offers", id]);
    // إزالة الـoffer فوراً من الـcache
    queryClient.setQueryData(["sub-offers", id], (old) =>
      (old as SubOffer[])?.filter(o => o.id !== subOfferId)
    );
    return { previous };
  },
  onError: (_err, _vars, ctx) => {
    // rollback لو فَشل
    if (ctx?.previous) queryClient.setQueryData(["sub-offers", id], ctx.previous);
  },
  onSettled: () => {
    queryClient.invalidateQueries({ queryKey: ["sub-offers", id] });
  },
});
