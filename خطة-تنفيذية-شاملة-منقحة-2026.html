export function useDriverOffersSSE(
  { enabled = false }: { enabled?: boolean } = {}
) {
  const [offers, setOffers] = useState<DriverOffer[]>([]);
  const [connected, setConnected] = useState(false);

  const connect = useCallback(async () => {
    if (!enabled) return;
    const token = await getSseToken()
      .catch(() => getAccessToken() ?? '');
    const url = `/sse/drivers/me/offers?token=${
      encodeURIComponent(token ?? '')}`;
    const es = new EventSource(url);
    esRef.current = es;
    es.addEventListener('offer', (e) => {
      try { setOffers(JSON.parse(e.data)); } catch {}
    });
    es.onopen = () => setConnected(true);
    es.onerror = () => { setConnected(false); es.close(); };
  }, [enabled]);

  useEffect(() => {
    connect();
    return () => { esRef.current?.close(); setConnected(false); };
  }, [connect]);

  return { offers, connected };
}
